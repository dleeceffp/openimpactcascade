# Click scripts for setting up a few things needed for running a cloud run container

# Create service account
export PROJECT_ID="oicsbx"
export SERVICE_NAME="dev-oicv21-cr-sa"
export SERVICE_ACCOUNT="dev-oicv21-cr-sa@$PROJECT_ID.iam.gserviceaccount.com"

gcloud iam service-accounts create $SERVICE_NAME \
  --display-name "OIC CR_Flask V21 Svc acct"

# 1.  Grant Vertex AI permissions
gcloud projects add-iam-policy-binding $PROJECT_ID \
  --member="serviceAccount:${SERVICE_ACCOUNT}" \
  --role="roles/aiplatform.user"

# 2. Cloud Storage (NEW - critical for writes)
gcloud projects add-iam-policy-binding $PROJECT_ID \
  --member="serviceAccount:${SERVICE_ACCOUNT}" \
  --role="roles/storage.objectAdmin"

# 3. Cloud Logging (NEW - recommended)
gcloud projects add-iam-policy-binding $PROJECT_ID \
  --member="serviceAccount:${SERVICE_ACCOUNT}" \
  --role="roles/logging.logWriter"

# 4. Secret Manager (NEW - for API keys)
# Auto-granted by Cloud Run with --set-secrets
# But can be set manually:
gcloud secrets add-iam-policy-binding ANTHROPIC_API_KEY \
  --member="serviceAccount:${SERVICE_ACCOUNT}" \
  --role="roles/secretmanager.secretAccessor"

gcloud secrets add-iam-policy-binding SECRET_KEY \
    --member="serviceAccount:${SERVICE_ACCOUNT}" \
    --role="roles/secretmanager.secretAccessor"


# Deploy with service account
gcloud run deploy $SERVICE_NAME \
  --image gcr.io/$PROJECT_ID/$SERVICE_NAME \
  --service-account risk-app-sa@$PROJECT_ID.iam.gserviceaccount.com \
  [... other flags ...]
