export ORG_ID="316939150015"
export PROJECT_ID="oicsbx"
export MEMBER="group:gcp-devopsml-admins@ymmvindustrial.com"

echo "Granting organization-level role to create new projects..."
gcloud organizations add-iam-policy-binding "${ORG_ID}" \
    --member="${MEMBER}" \
    --role="roles/resourcemanager.projectCreator" \
    --condition=None

echo "Granting project-level roles to project: ${PROJECT_ID} for member: ${MEMBER}..."

# Role to enable APIs
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/serviceusage.serviceUsageAdmin" \
    --condition=None

# Role to create service accounts
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/iam.serviceAccountAdmin" \
    --condition=None

# Role for Vertex AI access
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/aiplatform.user" \
    --condition=None

# Role for BigQuery access
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/bigquery.admin" \
    --condition=None

# Role for Cloud Build (building containers)
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/cloudbuild.editor" \
    --condition=None

# Role for Artifact Registry (storing container images)
gcloud projects add-iam-policy-binding "${PROJECT_ID}" \
    --member="${MEMBER}" \
    --role="roles/artifactregistry.admin" \
    --condition=None
